## Goal

---

이번 글의 목표는 ECC 라이브러리인 micro-ecc의 를 타겟으로 부채널분석을 통해 비밀 값 k를 구하는 것이 목표입니다.

타원곡선 상의 점인 P에 대해 Q = k \* P를 계산을 하는데 k의 비트 처리 부분에서 1과 0의 계산 차이를 통해 부채널 분석을 수행하게 됩니다.

uint8_t setreg(uint8_t* x, uint8_t len)
{
        uint32_t val;
        val = x[4] + (x[3] << 8) + (x[2] << 16) + (x[1] << 24);
// Must match capture/trace/TraceWhisperer.py:
//0:  DWT->CTRL
//1:  DWT->COMP0
//2:  DWT->COMP1
//3:  ETM->CR
//4:  ETM->TESSEICR
//5:  ETM->TEEVR
//6:  ETM->TECR1
//7:  ETM->TRACEIDR
//8:  TPI->ACPR
//9:  TPI->SPPR
//10: TPI->FFCR
//11: TPI->CSPSR
//12: ITM->TCR
        if       (x[0] == 0)    {DWT->CTRL = val;}
        else if  (x[0] == 1)    {DWT->COMP0 = val;}
        else if  (x[0] == 2)    {DWT->COMP1 = val;}
        else if  (x[0] == 3)    {ETM_SetupMode(); ETM->CR = val; ETM_TraceMode();}
        else if  (x[0] == 4)    {ETM_SetupMode(); ETM->TESSEICR = val; ETM_TraceMode();}
        else if  (x[0] == 5)    {ETM_SetupMode(); ETM->TEEVR    = val; ETM_TraceMode();}
        else if  (x[0] == 6)    {ETM_SetupMode(); ETM->TECR1    = val; ETM_TraceMode();}
        else if  (x[0] == 7)    {ETM_SetupMode(); ETM->TRACEIDR = val; ETM_TraceMode();}
        else if  (x[0] == 8)    {TPI->ACPR    = val;}
        else if  (x[0] == 9)    {TPI->SPPR    = val;}
        else if  (x[0] == 10)   {TPI->FFCR    = val;}
        else if  (x[0] == 11)   {TPI->CSPSR   = val;}
        else if  (x[0] == 12)   {ITM->TCR     = val;}

	return 0x00;
}

uint8_t getreg(uint8_t* x, uint8_t len)
{
        uint32_t val;
        if       (x[0] == 0)    {val = DWT->CTRL;}
        else if  (x[0] == 1)    {val = DWT->COMP0;}
        else if  (x[0] == 2)    {val = DWT->COMP1 ;}
        else if  (x[0] == 3)    {val = ETM->CR;}
        else if  (x[0] == 4)    {val = ETM->TESSEICR;}
        else if  (x[0] == 5)    {val = ETM->TEEVR;}
        else if  (x[0] == 6)    {val = ETM->TECR1;}
        else if  (x[0] == 7)    {val = ETM->TRACEIDR;}
        else if  (x[0] == 8)    {val = TPI->ACPR;}
        else if  (x[0] == 9)    {val = TPI->SPPR;}
        else if  (x[0] == 10)   {val = TPI->FFCR;}
        else if  (x[0] == 11)   {val = TPI->CSPSR;}
        else if  (x[0] == 12)   {val = ITM->TCR;}
        else {val = 0;}

        x[3] = val & 0xff;
        x[2] = (val >> 8) & 0xff;
        x[1] = (val >> 16) & 0xff;
        x[0] = (val >> 24) & 0xff;
	simpleserial_put('r', 4, x);
	return 0x00;
}

void enable_trace(void)
{
    // Enable SWO pin (not required on K82)
    #if (HAL_TYPE == HAL_stm32f3) || (HAL_TYPE == HAL_stm32f4)
       DBGMCU->CR |= DBGMCU_CR_TRACE_IOEN_Msk;
    #endif

    // Configure TPI
    CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk; // Enable access to registers
    TPI->ACPR = 0; // SWO trace baud rate = cpu clock / (ACPR+1)

    #if (HAL_TYPE == HAL_stm32f3) || (HAL_TYPE == HAL_stm32f4)
       TPI->SPPR = 2; // default to SWO with NRZ encoding
       //TPI->SPPR = 1; // SWO with Manchester encoding
    #else
       TPI->SPPR = 0; // default to parallel trace mode
    #endif

    TPI->FFCR = 0x102; // packet framing enabled
    //TPI->FFCR = 0x100; // no framing: for DWT/ITM only, no ETM
    TPI->CSPSR =0x00000008; // 4 trace lanes

    // Configure ITM:
    ITM->LAR = 0xC5ACCE55;
    ITM->TCR = (1 << ITM_TCR_TraceBusID_Pos) // Trace bus ID for TPIU
             | (1 << ITM_TCR_DWTENA_Pos) // Enable events from DWT
             | (0 << ITM_TCR_SYNCENA_Pos) // Disable sync packets!
             | (1 << ITM_TCR_ITMENA_Pos); // Main enable for ITM
    ITM->TER = 0xFFFFFFFF; // Enable all stimulus ports
    ITM->TPR = 0x00000000; // allow unpriviledged access

    // Configure DWT:
    DWT->CTRL = (0xf << DWT_CTRL_POSTINIT_Pos);// countdown counter for PC sampling, must be written
                                               // before enabling PC sampling
    DWT->CTRL |=(1 << DWT_CTRL_CYCTAP_Pos)     // Prescaler for PC sampling: 0 = x32, 1 = x512
              | (8 << DWT_CTRL_POSTPRESET_Pos) // PC sampling postscaler
              | (0 << DWT_CTRL_PCSAMPLENA_Pos) // disable PC sampling
              | (1 << DWT_CTRL_SYNCTAP_Pos)    // sync packets every 16M cycles
              | (0 << DWT_CTRL_EXCTRCENA_Pos)  // disable exception trace
              | (1 << DWT_CTRL_CYCCNTENA_Pos); // enable cycle counter

    // Configure DWT PC comparator 0:
    DWT->COMP0 = 0x00001d60; // AES subbytes
    DWT->MASK0 = 0;
    DWT->FUNCTION0 = (0 << DWT_FUNCTION_DATAVMATCH_Pos) // address match
                   | (0 << DWT_FUNCTION_CYCMATCH_Pos)
                   | (0 << DWT_FUNCTION_EMITRANGE_Pos)
                   | (8 << DWT_FUNCTION_FUNCTION_Pos); // Iaddr CMPMATCH event

    // Configure DWT PC comparator 1:
    DWT->COMP1 = 0x00001d68; // AES mixcolumns
    DWT->MASK1 = 0;
    DWT->FUNCTION1 = (0 << DWT_FUNCTION_DATAVMATCH_Pos) // address match
                   | (0 << DWT_FUNCTION_CYCMATCH_Pos)
                   | (0 << DWT_FUNCTION_EMITRANGE_Pos)
                   | (8 << DWT_FUNCTION_FUNCTION_Pos); // Iaddr CMPMATCH event

    // Configure ETM:
    ETM->LAR = 0xC5ACCE55;
    ETM_SetupMode();
    ETM->CR = ETM_CR_ETMEN; // Enable ETM output port
    ETM->TRACEIDR = 1; // Trace bus ID for TPIU
    ETM->FFLR = 0; // Stall processor when FIFO is full
    ETM->TEEVR = 0x000150a0;    // EmbeddedICE comparator 0 or 1 (DWT->COMP0 or DWT->COMP1)
    //ETM->TEEVR = 0x00000020;    // EmbeddedICE comparator 0 only
    //ETM->TEEVR = 0x00000021;    // EmbeddedICE comparator 1 only
    ETM->TESSEICR = 0xf; // set EmbeddedICE watchpoint 0 as a TraceEnable start resource.
    ETM->TECR1 = 0; // tracing is unaffected by the trace start/stop logic
    ETM_TraceMode();
}

uint8_t set_pcsample_params(uint8_t* x, uint8_t len)
{
    uint8_t postinit;
    uint8_t postreset;
    uint8_t cyctap;
    pcsamp_enable = x[0] & 1;
    cyctap = x[1] & 1;
    postinit  = x[2] & 0xf;
    postreset = x[3] & 0xf;

    // must clear everything before updating postinit field:
    DWT->CTRL = 0;
    // then update postinit:
    DWT->CTRL = (postinit << DWT_CTRL_POSTINIT_Pos);
    // then update the reset, but don't turn on PC sampling yet;
    // that will be handled in trigger_high_pcsamp
    DWT->CTRL = (cyctap << DWT_CTRL_CYCTAP_Pos)
              | (postreset << DWT_CTRL_POSTPRESET_Pos)
              | (postinit << DWT_CTRL_POSTINIT_Pos)
              | (1 << DWT_CTRL_SYNCTAP_Pos)
              | (1 << DWT_CTRL_CYCCNTENA_Pos);
    simpleserial_put('r', 4, x);
    return 0x00;
}

// in order for PC sample packets to be easily parsed, PC sampling must
// begin *after* we start capturing trace data
void trigger_high_pcsamp(void)
{
    if (pcsamp_enable == 1)
    {
        DWT->CTRL |= (1 << DWT_CTRL_PCSAMPLENA_Pos); // enable PC sampling
    }
    trigger_high();
}

void trigger_low_pcsamp(void)
{
    trigger_low();
    DWT->CTRL &= ~(1 << DWT_CTRL_PCSAMPLENA_Pos); // disable PC sampling
}

// Use globals for pmul input (P) and output (Q) points because
// they're too big to transmit all in one simpleserial transfer:
uECC_word_t P[uECC_MAX_WORDS * 2];
uECC_word_t Q[uECC_MAX_WORDS * 2];

void print(const char *ptr)
{
   while (*ptr != (char)0) {
      putch(*ptr);
      ptr++;
   }
}

uint8_t set_px(uint8_t* x, uint8_t len)
{
   int i, j;
   for (i = 0; i < 8; i++) {
      P[7-i] = 0;
      for (j = 0; j < 4; j++) {
         P[7-i] |= x[i*4+j] << ((3-j)*8);
      }
   }
   return 0x00;
}

uint8_t set_py(uint8_t* y, uint8_t len)
{
   // TODO: this conversion is used in multiple places, move it to a function:
   int i, j;
   for (i = 0; i < 8; i++) {
      P[15-i] = 0;
      for (j = 0; j < 4; j++) {
         P[15-i] |= y[i*4+j] << ((3-j)*8);
      }
   }
   return 0x00;
}

uint8_t get_qx(uint8_t* x, uint8_t len)
{
    int i, j;
    for (i = 0; i < 8; i++) {
       for (j = 0; j < 4; j++) {
          x[i*4+j] = (int)((Q[7-i] >> (3-j)*8) & 255);
       }
    }
    simpleserial_put('r', 32, x);
    return 0x00;
}

uint8_t get_qy(uint8_t* y, uint8_t len)
{
    int i, j;
    for (i = 0; i < 8; i++) {
       for (j = 0; j < 4; j++) {
          y[i*4+j] = (int)((Q[15-i] >> (3-j)*8) & 255);
       }
    }
    simpleserial_put('r', 32, y);
    return 0x00;
}

uint8_t run_pmul(uint8_t* k, uint8_t len)
{
    const struct uECC_Curve_t * curve;
    uECC_word_t kwords[uECC_MAX_WORDS];
    curve = uECC_secp256r1();

    int i, j;
    for (i = 0; i < 8; i++) {
       kwords[7-i] = 0;
       for (j = 0; j < 4; j++) {
          kwords[7-i] |= k[i*4+j] << ((3-j)*8);
       }
    }

    trigger_high_pcsamp();
    uECC_point_mult(Q, P, kwords, curve);
    trigger_low_pcsamp();
    return 0x00;
}

uint8_t run_pmul_fixed(uint8_t* k, uint8_t len)
{
    const struct uECC_Curve_t * curve;
    uECC_word_t kwords[uECC_MAX_WORDS];
    curve = uECC_secp256r1();

    int i, j;
    for (i = 0; i < 8; i++) {
       kwords[7-i] = 0;
       for (j = 0; j < 4; j++) {
          kwords[7-i] |= k[i*4+j] << ((3-j)*8);
       }
    }

    trigger_high_pcsamp();
    uECC_point_mult(Q, curve->G, kwords, curve);
    trigger_low_pcsamp();
    return 0x00;
}

uint8_t reset(uint8_t* x, uint8_t len)
{
    // Reset key here if needed
	return 0x00;
}

uint8_t info(uint8_t* x, uint8_t len)
{
        print("ChipWhisperer simpleserial-trace-ecc, compiled ");
        print(__DATE__);
        print(", ");
        print(__TIME__);
        print("\n");
	return 0x00;
}

uint8_t reenable_trace(uint8_t* x, uint8_t len)
{
        enable_trace();
	return 0x00;
}

int main(void)
{
    platform_init();
    init_uart();
    trigger_setup();

    simpleserial_init();
    simpleserial_addcmd('k', 32, run_pmul);
    simpleserial_addcmd('f', 32, run_pmul_fixed);
    simpleserial_addcmd('a', 32, set_px);
    simpleserial_addcmd('b', 32, set_py);
    simpleserial_addcmd('p', 32, get_qx);
    simpleserial_addcmd('q', 32, get_qy);
    simpleserial_addcmd('x',  0, reset);
    simpleserial_addcmd('i',  0, info);
    simpleserial_addcmd('e', 0, reenable_trace);
    simpleserial_addcmd('s', 5, setreg);
    simpleserial_addcmd('g', 5, getreg);
    simpleserial_addcmd('c', 4, set_pcsample_params);

    enable_trace();

    while(1)
        simpleserial_get();
}

## Trigger

---

전력 캡쳐 트리거의 경우 micro-ecc 라이브러리에서 ecc 연산의 비밀 값인 k의 비트를 연산하는 부분( [micro-ecc : EccPoint_mult()](https://github.com/kmackay/micro-ecc/blob/24c60e243580c7868f4334a1ba3123481fe1aa48/uECC.c#L862) )에 트리거 함수를 넣어서 k연산의 전력 파형을 캡쳐 할 수 있도록 했습니다.

먼저 펌웨어에는 아래와 같이 비밀 값 k를 세팅을 해줄 수 있는 부분이 있습니다. / simpleserial_addcmd('k', 32, run_pmul);

k는 총 256 bit의 값을 가질 수 있는데 0과 1의 전력 차이를 구하기 위해 다음 값으로 세팅을 하겠습니다.

```python
# big block of 1's, big block of 0's:
k = 0xf0000000fffffffefffffffffffffff04319055258e8617b0c46353d039cdaaf
kr = regularized_k(k)
hex(kr)
# kr = 0xeffffffffffffffffffffffffffffffff000000000000000000000000000000000000000
```

비밀 값 k는 실제 함수 동작에서 입력 되었을때 정규화 함수 regularized_k(k) 를 거쳐 사용되게 됩니다. 그래서 위 코드에서는 실제 연산에 사용되는 k를 세팅해야 하기 때문에 정규화 함수를 사용하여 비트가 1, 0으로 반반 나눠진 kr이 되도록 하는 k를 사용합니다.

*(사실은 위 코드에서 kr이 가지게 되는 값은 반반이긴 하지만 1비트가 모여있는 절반부분에서는 0이 하나가 있습니다, 근데 하나밖에 없어 사소하니까 괜찮아요.)*

또한 정규화된 kr은 256비트를 가지게 됩니다!  정규화 함수와 복원 코드는 아래와 같습니다.

```python
from ecpy.curves import Curve, Point
curve = Curve.get_curve('NIST-P256')

def regularized_k(input_k, bits=256):
    """Given input k, return the regularized k that the target processes (which the attack will retrieve).
    """
    assert input_k < curve.order
    kr = input_k + curve.order
    if kr & 2**bits:
        kr -= 2**bits
    return kr  

def input_k(kr, bits=256):
    """Given the regularized k that the target processes (which the attack will retrieve), return the regularized k that the target will be processing.
    """
    if kr < curve.order:
        kr += 2**bits
    i_k = kr - curve.order
    assert i_k < curve.order # sanity check
    return i_k
```

k를 0비트인 부분 128, 1비트인 부분 128개로 될 수 있도록 설정하신 것을 기억하시나요?

트리거당 캡쳐 부분을 255개의 세그먼트로 나누어 각 비트별로 전력이 얼마나 소모되는지 측정할겁니다.

여기서 256이 아닌 255인 이유는 트리거를 설정한 부분을 보시면 255회의 루프를 돌기  때문입니다.

정규화된 k는 256비트를 가진다 위에서 말했는데, 최하위 비트는 반복문 이후에 처리가 되게 됨으로 반복문이 돌아가는 횟수인 255를 세그먼트 크기로 세팅하게 됩니다.

실제로 정규화된 k 256 비트중 공격을 통해 알 수 있는 비트는 최하위를 제외한 중간 비트입니다.

각 비트별로 전력 트레이스를 확인하기 위해 세그먼트를 255개로 설정하고, 각 세그먼트당 170개의 샘플을 기록하도록 설정했습니다.

이제 캡쳐한 전력 트레이스를 바탕으로 0, 1 비트 부분의 전력 소모량을 분석하기 위해 먼저 전체 세그먼트의 평균을 구합니다.

```python
samples = scope.adc.samples  

avg_trace = np.zeros(samples)  

# 세그먼트들의 평균을 구하는 코드
for t in rtraces:  
    for i in range(1,255):  
        avg_trace += t.wave[i*samples:(i+1)*samples]  

avg_trace /= (255*len(rtraces)) # 트레이스 30개, 세그먼트 255개의 평균
```

k를 0, 1 비트 부분을 절반씩 나누어 설정 했기 때문에 전력 트레이스를 절반 나눠서 0 비트의 전력 소모 평균과, 1 비트의 전력 소모 평균을 구할 수 있습니다.

```python
avg_ones = np.zeros(samples)    # 1비트 전력 소모 평균 저장 변수
avg_zeros = np.zeros(samples)   # 0비트 전력 소모 평균 저장 변수

for t in rtraces:
    for i in range(128):
        # 세그먼트 255중 앞 부분은 정규화 k의 1비트 부분이므로 전력 트레이스(255*170) 에서
        # 1비트 부분인 t.wave[0:128*170]을 각 비트 한개씩해서 계산
        avg_ones += t.wave[i*samples:(i+1)*samples]           # 1비트 전력 소모량 총 합 계산
        # 아래는 0비트
        avg_zeros += t.wave[(i+127)*samples:(i+128)*samples]  # 0베트 전력 소모량 총 합 계산

avg_ones /= (127*len(rtraces))    # 1비트 전력 소모량 평균 계산
avg_zeros /= (127*len(rtraces))   # 0비트 전력 소모량 평균 계산
```

```python
# 수집한 트레이스에서 지정한 poi 위치의 전력 소모량 평균을 내는 함수
def calc_sumdata(poi, ptraces, trim=None):
    if trim:
        samples = trim
    else:
        samples = scope.adc.samples
    sumdata = np.zeros(255)
    for i in range(255):
        for t in ptraces:
            for p in poi:
                sample = t.wave[i*samples+abs(p)]
                if p >= 0:
                    sumdata[i] += sample
                else:
                    sumdata[i] -= sample
    return sumdata/len(ptraces)
```

파란색 그래프가 0, 1 비트가 절반으로 나눠진 k를 사용한 그래프이고, 빨간색 그래프는 0, 1이 잘 섞여 있는 그래프입니다.

파란색 그래프의 경우 대략 128 오프셋 기준으로 그래프가 위, 아래로 구분이 되어 있는데,

이는 저희가 k의 값을 0~127 : 1 bit, 128~ 255 : 0 bit 로 설정 했기 때문에 분명한 차이가 보인다는 것을 알 수 있습니다.

반면 빨간색 그래프의 경우 잘 섞여있는 k를 사용했기 대문에 차이가 매우 작음을 알 수 있습니다.

## Attack

---

프로파일링을 통해 micro-ecc에서 점 곱을 할 때, 0, 1 비트의 전력 차이와 누출 지점(poi)를 알게 되었습니다.

이제 이를 통해 랜덤한 k를 사용하여 k를 복구하는 공격을 해보겠습니다.

먼저 누출 지점이 아닌 앞 부분은 필요가 없어서 오프셋을 누출 시작점(START)으로 설정하고, poi도 설정한 오프셋에 맞추어 값을 바꿔 줍니다.

```python
scope.adc.samples = 170
scope.adc.offset = START # 90

# adjust POI accordingly:
for i in range(len(poi)):
    if poi[i] > 0:
        poi[i] -= scope.adc.offset
    else:
        poi[i] += scope.adc.offset
```

이제 랜덤한 k를 뽑아서 공격이 충분히 잘 성공할 수 있도록 전력 트레이스를 200개 수집해줍니다.

k를 랜덤하게 골랐기 때문에 0, 1 비트가 섞여 그래프가 위 아래로 어느정도 불규칙적으로 반복되는 모습을 볼 수 있습니다.

공격 코드는 아래와 같습니다.

 calc_sumdata(poi, straces, trim=trim) 함수를 통해 구한 sd를 통해 sd 값의 평균보다 크면 1 bit, 작으면 0 bit 로 해서 공격을 수행하게 됩니다. 이때 첫 비트는 펌웨어에서 255번 반복문 이전의 연산에 의해 전력이 올바르게 수집되지 않는 경우 때문에 전력 분석으로 구하지 않게 되고, 마지막 비트는 반복문 밖에서의 연산으로 전력 수집을 하지 않아 모르기 때문에 4가지의 k를 만들어 실제 k와 같은지 확인합니다.

공격 성공은 추측한 비트에 한해서 값이 같은 경우 성공으로 간주할 것 입니다. (최상위, 최하위 비트는 전력 분석을 하지 않아 모르기 때문)

```python
def attack(poi, straces, trim=None, verbose=True):
    sd = calc_sumdata(poi, straces, trim=trim)   # POI로 각 비트(255개)의 1/0 점수 계산

    # 가운데 비트(1~254) 판정
    # 0번째 비트를 추측을 안하는 이유는 펌웨어 255반복 이전의 연산의 이해 전력 안정화가 안되어
    # 잘못된 수집이 될 수도 있어서 그렇다네요~ 그냥 0 or 1 로 추측하는게 값 싸서 그런가 봅니다.
    guess = ''
    for i in range(1,255):                        # 세그먼트 1~254
        if sd[i] > np.average(sd):                # 점수가 전체 평균보다 크면
            guess += '1'                          #   1 bit
        else:
            guess += '0'                          #   아니면 0 bit

    # 첫 비트, 마지막 비트는 모르기 때문에 0, 1 bit 전부 시도한 총 4가지 경우의 k값 생성
    guesses = []
    for first in (['0', '1']):
        for last in (['0', '1']):
            guesses.append(int(first + guess + last, 2))  # 앞 + 가운데 + 뒤 → 정수 k 후보 4개

    kr = regularized_k(k)                          # k 정규화 => 실제 점곱에서 사용된 비밀 값
    wrong_bits = []
    if kr in guesses:                              # 후보 4개 중 정답이 있으면
        if verbose: print('✅ Guessed right!')     # 공격 성공
    else:                                          # 없으면 실패, 틀린 비트 집계
        for kbit in range(1,254):
            if int(guess[kbit-1]) != ((kr >> (255-kbit)) & 1):  # 추측 vs 정답 비트 비교
                wrong_bits.append(255-kbit)        #   다르면 틀린 비트 위치 기록
        if verbose:
            print('Attack failed.')
            print('Guesses: %s' % hex(guesses[0])) # 후보 4개, 정답 출력
            ...
            print('%d wrong bits' % len(wrong_bits))
    return wrong_bits                              # 틀린 비트 리스트 (성공하면 빈 리스트)
```

## Conclusion and Next

---

micro-ecc 라이브러리의 secp256r1 curve에 대하여 펌웨어에 트리거를 설정하여 부채널 분석을 수행 함으로, 비밀 논스 값인 k를 구할 수 있음을 알아봤는데요. 이번 글의 경우 펌웨어에 트리거를 설정해서 쉽게 부채널 분석을 수행했었습니다. 하지만 실제 기기에는 이런식으로는 하기가 쉽지 않은데요. 그래서 다음 글은 **"****소프트웨어 ECC 부채널 분석 해보기 - 2 (SAD를 사용한 부채널 분석)"** 으로 펌웨어와 타겟은 동일하나, 펌웨어 트리거 설정이 아닌 SAD(절대차 합, Sum of Absolute Differences)를 통해 부채널 분석을 진행하는 글로 찾아오겠습니다.
